import {spawn, spawnSync} from 'node:child_process';
import {createHash} from 'node:crypto';
import {createWriteStream, writeFileSync, mkdirSync, existsSync,readFileSync} from 'node:fs';
import {resolve} from 'node:path';
const [label, command, ...args] = process.argv.slice(2);
if (!/^[a-z0-9-]+$/.test(label) || !command) throw new Error('label and command required');
if (process.env.LUMEN_RELEASE_BUILD) throw new Error('ordinary gates must be marker-free');
const directory = resolve(import.meta.dirname, 'sdd/2026-10-12-production-readiness/task14c-evidence');
mkdirSync(directory,{recursive:true});
if (existsSync(resolve(directory,`${label}.started.json`))) throw new Error('refusing overwrite');
const snapshot=spawnSync('powershell.exe',['-NoProfile','-File',resolve(import.meta.dirname,'task14c-host.ps1')],{encoding:'utf8',windowsHide:true});
writeFileSync(resolve(directory,`${label}.host.json`),snapshot.stdout);
writeFileSync(resolve(directory,`${label}.host.stderr.log`),snapshot.stderr);
if(snapshot.status!==0) throw new Error('host snapshot failed');
const head=spawnSync('rtk',['proxy','git','rev-parse','HEAD']).stdout.toString().trim();
const patch=spawnSync('rtk',['proxy','git','diff','--binary']).stdout;
writeFileSync(resolve(directory,`${label}.source.patch`),patch);
const base={label,command,args,cwd:process.cwd(),started:new Date().toISOString(),head,patchSha256:createHash('sha256').update(patch).digest('hex'),releaseMarker:process.env.LUMEN_RELEASE_BUILD??null,runnerPid:process.pid};
const stdout=createWriteStream(resolve(directory,`${label}.stdout.log`));
const stderr=createWriteStream(resolve(directory,`${label}.stderr.log`));
const child=spawn(command,args,{cwd:process.cwd(),env:process.env,windowsHide:true});
const receipt={...base,pid:child.pid};
const report=resolve(directory,'../task-14c-report.md');
const checkpoint=(message)=>writeFileSync(report,`# Latest gate checkpoint ${new Date().toISOString()}\n\n${message}\n\n${readFileSync(report,'utf8')}`);
checkpoint(`- START ${label}; command ${JSON.stringify([command,...args])}; logger PID ${process.pid}, command PID ${child.pid}; raw task14c-evidence/${label}.*; HEAD ${head}; patch ${base.patchSha256}; marker absent. Actual exit pending. Preflight host/process/port/volume JSON retained.`);
writeFileSync(resolve(directory,`${label}.started.json`),JSON.stringify(receipt,null,2));
console.log(JSON.stringify(receipt));
child.stdout.pipe(stdout);child.stderr.pipe(stderr);
child.on('error',error=>console.error(error));
child.on('close',(code,signal)=>{
 writeFileSync(resolve(directory,`${label}.json`),JSON.stringify({...receipt,ended:new Date().toISOString(),code,signal},null,2));
 checkpoint(`- FINISH ${label}; actual code ${code}, signal ${signal}; command PID ${child.pid}; raw and final receipt task14c-evidence/${label}.*. No gate inference beyond actual output.`);
 console.log(JSON.stringify({label,code,signal,pid:child.pid}));process.exitCode=code??1;
});
